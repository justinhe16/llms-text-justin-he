# Runtime dependencies for the backend image (FastAPI API + ARQ worker).
# Pinned exactly so the image CI builds is the image that was tested.

# Web framework
fastapi==0.141.1
uvicorn[standard]==0.52.1   # ASGI server; [standard] adds uvloop and httptools

# Configuration
pydantic-settings==2.14.2   # environment parsing and validation for app/core/settings.py

# Database
asyncpg==0.31.0   # Postgres driver used by app/infrastructure/db/ — see requirements-dev.txt
                   # for why asyncpg-stubs is a separate, dev-only dependency

# Auth
pyjwt[crypto]==2.13.0   # local JWT verification against the Supabase JWKS (app/core/auth/)
cryptography==50.0.0    # pinned on its own line even though pyjwt[crypto] pulls it in: an
                         # extra alone floats the transitive pin, and this file's policy is
                         # "the image CI builds is the image that was tested"

# Job queue
arq==0.28.0     # ARQ worker (app/worker/) and the API's enqueue pool. RUNTIME for BOTH
                # processes, not just the worker: app/main.py's lifespan opens an ArqRedis
                # pool so routers can enqueue, so the API image needs it too.
redis==5.3.1    # pinned on its own line even though arq depends on `redis[hiredis]<6,>=4.2.0`:
                # that range would let a `pip install` months apart produce two different
                # images from one commit, and this file's policy is "the image CI builds is
                # the image that was tested". Must stay <6 while arq's pin says so.
hiredis==3.4.0  # the C parser behind redis's [hiredis] extra, pinned for the same reason
                # cryptography is below — an extra alone floats the transitive pin

# HTTP client
httpx==0.28.1   # RUNTIME, not dev: app/core/auth/jwks.py fetches the JWKS document with it.
                # build-check installs this file alone, so an httpx that lived only in
                # requirements-dev.txt would pass lint and test and fail there — and only
                # there. The test suite's ASGITransport client inherits it from here.

# Content extraction
trafilatura==2.2.0   # RUNTIME for the WORKER process only — the one entry here that is. The
                     # API never parses HTML; only the crawl task reaches
                     # app/features/crawl/internals/extract.py. Compare arq and httpx above,
                     # whose comments explain why they are runtime for BOTH processes: this
                     # one is in this file because the worker and the API are one image
                     # (CLAUDE.md "Stack"), not because the API imports it.
                     #
                     # It arrives with a dependency tree of its own — lxml, jusText, courlan,
                     # htmldate, and those packages' own dependencies (babel, dateparser, tld,
                     # regex, ...). Those four are what decide the text that comes out of a
                     # page, so each gets its own pinned line below, as does lxml_html_clean,
                     # which reaches the image only through an extra. The rest are left to
                     # resolve: pinning a dependency-of-a-dependency this file never names
                     # would freeze a graph nobody here is reading, and fastapi and uvicorn
                     # above do not do that either.
lxml==6.1.1          # pinned on its own line even though trafilatura requires `lxml>=6.1.1`:
                     # a floor with no ceiling would let a `pip install` months apart resolve
                     # two different parsers from one commit. Same argument as redis above,
                     # with a sharper point — a floated redis changes a version string, a
                     # floated lxml can change the extracted body itself, which is what the
                     # fixtures under tests/fixtures/ assert against.
lxml_html_clean==0.4.5  # pinned for exactly the reason cryptography and hiredis are: it is
                        # reachable only through jusText's `lxml[html_clean]` extra, and an
                        # extra alone floats the transitive pin.
jusText==3.0.2       # one of the fallbacks trafilatura's cascade drops to when its own XPath
                     # pass finds nothing, so it is load-bearing for exactly the pages that
                     # are hardest to extract. `justext>=3.0.2` upstream — floor only, same
                     # risk as lxml. Spelled as PyPI spells it; pip normalizes the case.
courlan==1.4.0       # trafilatura's URL cleaning, imported at its import time. `>=1.4.0`
                     # upstream, floor only.
htmldate==1.10.0     # trafilatura's date extraction, imported at its import time. `>=1.10.0`
                     # upstream, floor only. extract.py discards the date and passes
                     # `extensive=False` to stop it hunting for one, but the import still has
                     # to resolve, so the version still has to be fixed.
